file(GLOB_RECURSE SOURCE_FILES CONFIGURE_DEPENDS "*.cpp")

add_library(core SHARED ${SOURCE_FILES})
target_link_libraries(core PUBLIC ngx ${JNI_LIBRARIES} VulkanMemoryAllocator volk::volk glfw streamline)
if (MCVR_ENABLE_OMM)
    target_link_libraries(core PUBLIC omm-lib)
    target_compile_definitions(core PUBLIC MCVR_ENABLE_OMM)
endif ()
# NVIDIA Aftermath SDK for GPU crash dump collection (headers only — DLL loaded at runtime)
set(AFTERMATH_SDK_DIR "${CMAKE_SOURCE_DIR}/extern/aftermath")
# miniz for Anvil region file zlib decompression (from OMM tree, export stub in anvil/)
set(MINIZ_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/extern/omm/external/donut/thirdparty/miniz")
set(ANVIL_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/render/anvil")
set(NLOHMANN_JSON_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/extern/json/include")
set(MINIZIP_NG_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/extern/minizip-ng")
set(TINYEXPR_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/extern/tinyexpr")
target_include_directories(core PUBLIC ${GENTOOL_OUTPUT_DIR} ${VOLK_INCLUDE_DIR} ${GLFW_INCLUDE_DIR} ${VMA_INCLUDE_DIR} ${STB_INCLUDE_DIR} ${JNI_INCLUDE_DIRS} ${GLM_INCLUDE_DIR} ${AFTERMATH_SDK_DIR}/include ${MINIZ_INCLUDE_DIR} ${ANVIL_INCLUDE_DIR} ${NLOHMANN_JSON_INCLUDE_DIR} ${MINIZIP_NG_INCLUDE_DIR} ${TINYEXPR_INCLUDE_DIR})
target_compile_definitions(core PUBLIC VK_NO_PROTOTYPES CORE_LIB)
if (USE_AMD)
    target_compile_definitions(core PUBLIC USE_AMD)
endif ()
if (MCVR_ENABLE_FFX_UPSCALER)
    # amd_fidelityfx_vk provides the main FFX API (ffxCreateContext, ffxQuery, etc.)
    target_link_libraries(core PUBLIC amd_fidelityfx_vk ffx_backend_vk_x64 ffx_fsr3upscaler_x64)
    target_compile_definitions(core PUBLIC MCVR_ENABLE_FFX_UPSCALER)
endif ()
if (MCVR_ENABLE_SHARC)
    target_compile_definitions(core PUBLIC MCVR_ENABLE_SHARC)
    target_compile_definitions(core PUBLIC MCVR_SHARC_MAIN_TRACE_QUERY_MODE=1)
endif ()
if (MCVR_ENABLE_NRD)
    target_link_libraries(core PUBLIC NRD)
    target_include_directories(core PUBLIC ${NRD_INCLUDE_DIR})
    target_compile_definitions(core PUBLIC NRD_STATIC_LIBRARY=1)
    target_compile_definitions(core PUBLIC MCVR_ENABLE_NRD)
endif ()
if (MSVC)
    target_compile_definitions(core PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(core PRIVATE /utf-8)
    target_link_libraries(core PRIVATE d3d11 dxgi dcomp dwmapi)
endif ()

if (MCVR_ENABLE_ENGINE_V2)
    target_link_libraries(core PUBLIC engine)
    target_compile_definitions(core PUBLIC MCVR_ENABLE_ENGINE_V2)
endif ()

set_target_properties(core PROPERTIES CXX_VISIBILITY_PRESET default VISIBILITY_INLINES_HIDDEN OFF)

install(TARGETS core DESTINATION ${MCVR_INSTALL_LIB_DIR})
install(TARGETS core DESTINATION ${MCVR_INSTALL_BIN_DIR})
if (MSVC)
    install(FILES $<TARGET_PDB_FILE:core> DESTINATION ${MCVR_INSTALL_LIB_DIR} OPTIONAL)
    install(FILES $<TARGET_PDB_FILE:core> DESTINATION ${MCVR_INSTALL_BIN_DIR} OPTIONAL)
endif ()
